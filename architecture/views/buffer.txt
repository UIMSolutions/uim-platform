  script
    const modal = document.getElementById('abb-modal');
    const tableBody = document.getElementById('abb-table-body');

    // Event Delegation: Listen for clicks on table rows
    tableBody.addEventListener('click', (event) => {
      const row = event.target.closest('tr');
      if (!row) return;

      const cells = row.cells;

      // Extract content from clicked row
      document.getElementById('modal-date').textContent = cells[0].textContent;
      document.getElementById('modal-title').textContent = cells[1].textContent;
      document.getElementById('modal-layer').textContent = cells[2].textContent;
      document.getElementById('modal-purpose').textContent = cells[3].textContent;
      document.getElementById('modal-tech').textContent = cells[4].textContent;
      document.getElementById('modal-status').innerHTML = cells[5].innerHTML;
      document.getElementById('modal-desc').textContent = row.getAttribute('data-desc') || 'No additional architectural description provided.';

      // Native browser showModal method (Pico styles this automatically)
      modal.showModal();
    });

    // Close Modal Handler
    const closeModal = () => modal.close();
    document.getElementById('close-modal-btn').addEventListener('click', closeModal);
    document.getElementById('close-modal-footer').addEventListener('click', closeModal);

    // Close when clicking outside the modal dialog box
    modal.addEventListener('click', (event) => {
      if (event.target === modal) closeModal();
    });
